function generate_corn_data()
disp("🌱 Generating corn_data.csv ...");
rows = 14; cols = 21;
fid = fopen('data/corn_data.csv','w');
fprintf(fid,'timestamp,row,col,NIR,Red,latitude,longitude,zone\n');

for r = 1:rows
    for c = 1:cols
        tstamp = datetime('now') - minutes(r*c);
        NIR = 0.5 + 0.3*rand();
        Red = 0.2 + 0.2*rand();
        lat = 26.9 + 0.001*(r-7);
        lon = 75.8 + 0.001*(c-10);
        zone = strcat('Zone-', string(ceil(c/7)));
        fprintf(fid, '%s,%d,%d,%.3f,%.3f,%.6f,%.6f,%s\n', ...
            char(datetime(tstamp,'Format','yyyy-MM-dd HH:mm:ss')), r, c, NIR, Red, lat, lon, zone);
    end
end
fclose(fid);
disp("✅ corn_data.csv generated successfully.");
end
