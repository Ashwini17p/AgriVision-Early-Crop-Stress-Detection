function compareCropWaterNeeds()
    cropDB = createCropDatabase();
    
    % Standard conditions
    baseConditions = struct();
    baseConditions.growthStage = 'Mid-season';
    baseConditions.soilMoisture = 25;
    baseConditions.fieldCapacity = 35;
    baseConditions.wiltingPoint = 15;
    baseConditions.temperature = 28;
    baseConditions.humidity = 65;
    baseConditions.rainfall_last7days = 0;
    baseConditions.rainfall_forecast = zeros(1,7);
    baseConditions.windSpeed = 2.0;
    baseConditions.solarRadiation = 22.0;
    baseConditions.ndvi = 0.65;
    baseConditions.cropHeight = 40;
    baseConditions.latitude = 18.5204;
    baseConditions.longitude = 73.8567;
    baseConditions.plantingDate = datetime('2025-06-15');
    baseConditions.daysAfterPlanting = 50;
    baseConditions.soilType = 'Loamy';
    
    crops = cropDB.Crop;
    waterNeeds = zeros(length(crops), 1);
    
    fprintf('Calculating water needs for all crops...\n\n');
    
    for i = 1:length(crops)
        baseConditions.cropType = crops{i};
        plan = calculateWaterRequirements(cropDB, baseConditions);
        waterNeeds(i) = plan.dailyWaterNeed_mm;
        fprintf('  %s: %.2f mm/day\n', crops{i}, waterNeeds(i));
    end
    
    % Visualize
    figure('Name', 'Crop Water Comparison', 'Position', [200, 200, 1000, 600]);
    
    [sorted, idx] = sort(waterNeeds, 'descend');
    sortedCrops = crops(idx);
    
    colors = zeros(length(sorted), 3);
    for i = 1:length(sorted)
        if i == 1
            colors(i,:) = [1 0.3 0.3];  % Highest in red
        elseif i == length(sorted)
            colors(i,:) = [0.2 0.8 0.2];  % Lowest in green
        else
            colors(i,:) = [0.3 0.6 1];  % Others in blue
        end
    end
    
    b = bar(sorted, 'FaceColor', 'flat');
    b.CData = colors;
    
    set(gca, 'XTickLabel', sortedCrops, 'FontWeight', 'bold', 'FontSize', 11);
    xtickangle(45);
    ylabel('Water Requirement (mm/day)', 'FontWeight', 'bold', 'FontSize', 12);
    title('Comparative Water Requirements (Mid-Season, Standard Conditions)', ...
          'FontWeight', 'bold', 'FontSize', 13);
    grid on;
    
    for i = 1:length(sorted)
        text(i, sorted(i) + 0.3, sprintf('%.1f', sorted(i)), ...
            'HorizontalAlignment', 'center', 'FontWeight', 'bold', 'FontSize', 10);
    end
    
    fprintf('\n✅ Comparison complete!\n');
    fprintf('Key Insights:\n');
    fprintf('  Highest: %s (%.2f mm/day)\n', sortedCrops{1}, sorted(1));
    fprintf('  Lowest: %s (%.2f mm/day)\n', sortedCrops{end}, sorted(end));
    fprintf('  Ratio: %.1fx difference\n\n', sorted(1)/sorted(end));
end

function analyzeDroughtScenario()
    fprintf('Simulating 30-day drought scenario for Cotton...\n\n');
    
    conditions = struct();
    conditions.cropType = 'Cotton';
    conditions.growthStage = 'Mid-season';
    conditions.soilType = 'Loamy';
    conditions.fieldCapacity = 35;
    conditions.wiltingPoint = 15;
    conditions.temperature = 32;
    conditions.humidity = 45;
    conditions.windSpeed = 3.5;
    conditions.solarRadiation = 25;
    conditions.ndvi = 0.70;
    conditions.cropHeight = 50;
    conditions.latitude = 18.5204;
    conditions.longitude = 73.8567;
    conditions.plantingDate = datetime('2025-06-01');
    conditions.daysAfterPlanting = 60;
    conditions.rainfall_last7days = 0;
    conditions.rainfall_forecast = zeros(1,7);
    
    cropDB = createCropDatabase();
    
    days = 30;
    soilMoisture = zeros(days, 1);
    waterNeeds = zeros(days, 1);
    ndviValues = zeros(days, 1);
    
    currentMoisture = 30;
    currentNDVI = 0.70;
    
    for d = 1:days
        conditions.soilMoisture = currentMoisture;
        conditions.ndvi = currentNDVI;
        
        plan = calculateWaterRequirements(cropDB, conditions);
        
        soilMoisture(d) = currentMoisture;
        waterNeeds(d) = plan.dailyWaterNeed_mm;
        ndviValues(d) = currentNDVI;
        
        moistureLoss = plan.ETc / 10;
        currentMoisture = max(conditions.wiltingPoint, currentMoisture - moistureLoss);
        
        if currentMoisture < 20
            currentNDVI = max(0.2, currentNDVI - 0.01);
        elseif currentMoisture < 25
            currentNDVI = max(0.3, currentNDVI - 0.005);
        end
    end
    
    % Visualize
    figure('Name', 'Drought Scenario Analysis', 'Position', [100, 100, 1200, 700]);
    
    subplot(2, 2, 1);
    plot(1:days, soilMoisture, '-b', 'LineWidth', 2.5);
    hold on;
    yline(15, '--r', 'Wilting Point', 'LineWidth', 2);
    yline(25, '--', 'Color', [1 0.6 0], 'LineWidth', 2);
    text(5, 26, 'Stress Threshold', 'FontSize', 9, 'Color', [1 0.6 0]);
    xlabel('Days', 'FontWeight', 'bold', 'FontSize', 11);
    ylabel('Soil Moisture (%)', 'FontWeight', 'bold', 'FontSize', 11);
    title('Soil Moisture Depletion', 'FontSize', 13, 'FontWeight', 'bold');
    grid on;
    xlim([1 30]);
    
    subplot(2, 2, 2);
    plot(1:days, ndviValues, '-g', 'LineWidth', 2.5);
    hold on;
    yline(0.5, '--r', 'Stress Threshold', 'LineWidth', 2);
    xlabel('Days', 'FontWeight', 'bold', 'FontSize', 11);
    ylabel('NDVI', 'FontWeight', 'bold', 'FontSize', 11);
    title('Crop Health Decline', 'FontSize', 13, 'FontWeight', 'bold');
    grid on;
    xlim([1 30]);
    
    subplot(2, 2, 3);
    area(1:days, waterNeeds, 'FaceColor', [0.4 0.7 1], 'EdgeColor', 'b', 'LineWidth', 1.5);
    xlabel('Days', 'FontWeight', 'bold', 'FontSize', 11);
    ylabel('Water Need (mm/day)', 'FontWeight', 'bold', 'FontSize', 11);
    title('Increasing Water Demand', 'FontSize', 13, 'FontWeight', 'bold');
    grid on;
    xlim([1 30]);
    
    subplot(2, 2, 4);
    cumWater = cumsum(waterNeeds);
    plot(1:days, cumWater, '-r', 'LineWidth', 2.5);
    xlabel('Days', 'FontWeight', 'bold', 'FontSize', 11);
    ylabel('Cumulative Water (mm)', 'FontWeight', 'bold', 'FontSize', 11);
    title(sprintf('Total Water Deficit: %.1f mm', cumWater(end)), ...
          'FontSize', 13, 'FontWeight', 'bold');
    grid on;
    xlim([1 30]);
    
    sgtitle('🌵 30-Day Drought Impact Analysis - Cotton', 'FontSize', 16, 'FontWeight', 'bold');
    
    wiltDay = find(soilMoisture <= 15, 1);
    stressDay = find(ndviValues < 0.5, 1);
    
    fprintf('═══════════════════════════════════════\n');
    fprintf('DROUGHT IMPACT SUMMARY\n');
    fprintf('═══════════════════════════════════════\n\n');
    fprintf('Critical Milestones:\n');
    if ~isempty(wiltDay)
        fprintf('  Day %d: Soil moisture reaches wilting point (%.1f%%)\n', wiltDay, soilMoisture(wiltDay));
    else
        fprintf('  Wilting point not reached in 30 days\n');
    end
    if ~isempty(stressDay)
        fprintf('  Day %d: NDVI drops below stress threshold (%.2f)\n', stressDay, ndviValues(stressDay));
    else
        fprintf('  NDVI stress threshold not reached\n');
    end
    fprintf('\nWater Balance:\n');
    fprintf('  Total water deficit: %.1f mm\n', cumWater(end));
    fprintf('  Equivalent volume: %.1f m³/hectare\n', cumWater(end)*10);
    fprintf('  Average daily need: %.2f mm\n', mean(waterNeeds));
    fprintf('\nCrop Health:\n');
    fprintf('  Initial NDVI: %.2f\n', ndviValues(1));
    fprintf('  Final NDVI: %.2f\n', ndviValues(end));
    fprintf('  Health decline: %.1f%%\n', (1 - ndviValues(end)/ndviValues(1))*100);
    fprintf('\n═══════════════════════════════════════\n\n');
end

function createWaterViz(plan, forecast, conditions)
    figure('Name', 'Water Prediction Results', 'Position', [200, 200, 1200, 500]);
    
    % Left plot: 7-day forecast
    subplot(1, 2, 1);
    yyaxis left
    bar(forecast.days, forecast.dailyNeeds, 'FaceColor', [0.3 0.6 1]);
    ylabel('Water Need (mm/day)', 'FontWeight', 'bold', 'FontSize', 11);
    ylim([0 max(forecast.dailyNeeds) * 1.3]);
    
    yyaxis right
    plot(forecast.days, forecast.soilMoisture, '-ro', 'LineWidth', 2.5, 'MarkerSize', 8);
    hold on;
    yline(conditions.wiltingPoint, '--r', 'Wilting', 'LineWidth', 1.5);
    yline(conditions.fieldCapacity, '--g', 'Capacity', 'LineWidth', 1.5);
    ylabel('Soil Moisture (%)', 'FontWeight', 'bold', 'FontSize', 11);
    ylim([conditions.wiltingPoint - 5, conditions.fieldCapacity + 5]);
    
    xlabel('Day', 'FontWeight', 'bold', 'FontSize', 11);
    title('7-Day Water Forecast', 'FontWeight', 'bold', 'FontSize', 13);
    legend('Daily Need', 'Soil Moisture', 'Location', 'best');
    grid on;
    
    % Right plot: Before/After irrigation
    subplot(1, 2, 2);
    categories = {'Current\nMoisture', 'After\nIrrigation'};
    afterIrrigation = min(conditions.fieldCapacity, ...
                          conditions.soilMoisture + plan.dailyWaterNeed_mm/10);
    values = [conditions.soilMoisture, afterIrrigation];
    
    colors = [0.2 0.7 0.9; 0.2 0.8 0.3];
    b = bar(values, 'FaceColor', 'flat');
    b.CData = colors;
    
    hold on;
    yline(conditions.wiltingPoint, '--r', 'Wilting Point', 'LineWidth', 2);
    yline(conditions.fieldCapacity, '--g', 'Field Capacity', 'LineWidth', 2);
    
    set(gca, 'XTickLabel', categories, 'FontWeight', 'bold', 'FontSize', 11);
    ylabel('Soil Moisture (%)', 'FontWeight', 'bold', 'FontSize', 11);
    title('Irrigation Impact', 'FontWeight', 'bold', 'FontSize', 13);
    ylim([0 max(values) + 10]);
    grid on;
    
    % Add value labels
    for i = 1:2
        text(i, values(i) + 1, sprintf('%.1f%%', values(i)), ...
            'HorizontalAlignment', 'center', 'FontWeight', 'bold', 'FontSize', 11);
    end
    
    sgtitle(sprintf('💧 Water Prediction: %s (%s)', conditions.cropType, conditions.growthStage), ...
            'FontSize', 15, 'FontWeight', 'bold');
end

function db = createCropDatabase()

                 'Tomato', 'Potato', 'Soybean', 'Groundnut', 'Millet'};
    waterNeeds = [7.5, 8.5, 9.0, 6.0; 3.0, 4.5, 6.0, 3.5; 4.0, 5.5, 7.5, 5.0; ...

                  3.5, 5.0, 6.0, 4.0; 3.0, 4.5, 6.5, 4.0; 3.5, 5.0, 6.0, 3.5; ...
                  2.5, 3.5, 5.0, 3.0];
    kc_values = [1.05, 1.10, 1.20, 0.90; 0.30, 0.70, 1.15, 0.40; ...
                 0.35, 0.70, 1.15, 0.70; 0.40, 0.75, 1.25, 0.75; ...
                 0.30, 0.80, 1.20, 0.60; 0.45, 0.75, 1.15, 0.80; ...

                 0.40, 0.70, 1.05, 0.60; 0.30, 0.70, 1.05, 0.55];
    rootDepth = [30, 100, 100, 150, 100, 60, 50, 80, 60, 120]';
    droughtTolerance = [1, 3, 3, 2, 3, 2, 2, 3, 3, 5]';
    
    db = table(cropNames', waterNeeds, kc_values, rootDepth, droughtTolerance, ...
        'VariableNames', {'Crop', 'WaterNeeds_mmday', 'Kc', 'RootDepth_cm', 'DroughtTolerance'});
end

function plan = calculateWaterRequirements(cropDB, conditions)
    cropIdx = find(strcmp(cropDB.Crop, conditions.cropType));
    cropData = cropDB(cropIdx, :);
    
    stageNames = {'Initial', 'Development', 'Mid-season', 'Late-season'};
    stageIdx = find(strcmp(stageNames, conditions.growthStage));
    
    baseWaterNeed = cropData.WaterNeeds_mmday(stageIdx);
    ET0 = 0.0135 * conditions.temperature * 20;
    Kc = cropData.Kc(stageIdx);
    ETc = Kc * ET0;
    
    availableWater = conditions.soilMoisture - conditions.wiltingPoint;
    maxAvailableWater = conditions.fieldCapacity - conditions.wiltingPoint;
    moistureRatio = availableWater / maxAvailableWater;
    
    stressFactor = 1.0 + max(0, 1 - moistureRatio) * 0.5;
    healthFactor = 1.0 + max(0, 0.7 - conditions.ndvi) * 0.3;
    
    dailyWaterNeed = baseWaterNeed * stressFactor * healthFactor;
    
    plan.dailyWaterNeed_mm = dailyWaterNeed;
    plan.dailyWaterNeed_Lm2 = dailyWaterNeed;
    plan.dailyWaterNeed_totalm3 = dailyWaterNeed * 10;
    plan.irrigationDuration_hours = dailyWaterNeed / 4;
    plan.ET0 = ET0;
    plan.ETc = ETc;
    plan.Kc = Kc;
    plan.stressFactor = stressFactor;
    plan.healthFactor = healthFactor;
    plan.moistureRatio = moistureRatio;
    
    if moistureRatio < 0.4
        plan.urgency = 'CRITICAL - Irrigate Immediately';
    elseif moistureRatio < 0.6
        plan.urgency = 'URGENT - Irrigate Today';
    elseif moistureRatio < 0.8
        plan.urgency = 'SCHEDULED - Within 24h';
    else
        plan.urgency = 'ADEQUATE - Monitor';
    end
end

function forecast = predictWeeklyWater(plan, conditions)
    forecast.days = 1:7;
    forecast.dailyNeeds = repmat(plan.dailyWaterNeed_mm, 7, 1);
    
    currentMoisture = conditions.soilMoisture;
    moistureValues = zeros(7, 1);
    
    for d = 1:7
        moistureIncrease = plan.dailyWaterNeed_mm / 10;
        moistureLoss = plan.ETc / 10;
        currentMoisture = currentMoisture + moistureIncrease - moistureLoss;
        currentMoisture = min(conditions.fieldCapacity, max(conditions.wiltingPoint, currentMoisture));
        moistureValues(d) = currentMoisture;
    end
    
    forecast.soilMoisture = moistureValues;
    forecast.rainfall = conditions.rainfall_forecast;
    forecast.totalWeeklyNeed = sum(forecast.dailyNeeds);
    forecast.cumulativeWater = cumsum(forecast.dailyNeeds);
end
