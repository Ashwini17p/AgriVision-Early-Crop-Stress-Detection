%% Quick LSTM Test - Standalone Version
clc; clear; close all;

fprintf('🧪 AgriVision LSTM Quick Test\n');
fprintf('════════════════════════════════════════\n\n');

%% Step 1: Check if model exists
if exist('models/ndvi_lstm_model.mat', 'file')
    fprintf('✅ Found existing model. Loading...\n');
    load('models/ndvi_lstm_model.mat', 'net');
else
    fprintf('⚠️ No trained model found.\n');
    fprintf('Please train the model first by running:\n');
    fprintf('   >> train_ndvi_lstm_fixed\n\n');
    
    response = input('Train model now? (y/n): ', 's');
    if lower(response) == 'y'
        % Run the training function directly
        train_ndvi_lstm_fixed();
        
        % Load the newly trained model
        load('models/ndvi_lstm_model.mat', 'net');
    else
        error('Model required. Exiting...');
    end
end

%% Step 2: Test predictions on different scenarios
fprintf('\n🔬 Testing LSTM on different crop scenarios...\n\n');

% Scenario 1: Declining health (drought stress)
fprintf('📉 Scenario 1: Declining Health (Drought Stress)\n');
testSeq1 = [
    0.65, 0.62, 0.58, 0.55, 0.52, 0.48;   % NDVI decreasing
    30,   28,   25,   23,   22,   20;     % Soil moisture dropping
    65,   64,   66,   65,   63,   62      % Humidity stable
];

prediction1 = classify(net, testSeq1);
fprintf('   📊 Input: NDVI 0.65→0.48, Moisture 30%%→20%%\n');
fprintf('   🎯 Prediction: %s\n', char(prediction1));
fprintf('   💡 Interpretation: Drought stress detected - irrigate immediately\n\n');

% Scenario 2: Improving health (recovery)
fprintf('📈 Scenario 2: Improving Health (Recovery)\n');

];

prediction2 = classify(net, testSeq2);
fprintf('   📊 Input: NDVI 0.45→0.65, Moisture 20%%→33%%\n');
fprintf('   🎯 Prediction: %s\n', char(prediction2));
fprintf('   💡 Interpretation: Plant recovering after irrigation\n\n');

% Scenario 3: Stable healthy plant

% Scenario 4: Critical decline (disease + drought)
